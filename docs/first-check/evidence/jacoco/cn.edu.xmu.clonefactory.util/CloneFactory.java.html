package cn.edu.xmu.clonefactory.util;

import com.cdwy.permission.dto.CreateUserRequest;
import com.cdwy.permission.dto.PermissionResponse;
import com.cdwy.permission.dto.RoleResponse;
import com.cdwy.permission.dto.SystemResponse;
import com.cdwy.permission.dto.UpdateUserRequest;
import com.cdwy.permission.dto.UserResponse;
import com.cdwy.permission.entity.BusinessSystem;
import com.cdwy.permission.entity.Permission;
import com.cdwy.permission.entity.Role;
import com.cdwy.permission.entity.UserAccount;

public final class CloneFactory {
  /**
   * Copy all fields from source to target
   * generated by @CopyFrom(UserAccount) in {UserResponse}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static UserResponse copy(UserResponse target, UserAccount source) {
    target.setId(source.getId());
    target.setAdministrator(source.getAdministrator());
    target.setUsername(source.getUsername());
    target.setDisplayName(source.getDisplayName());
    target.setEnabled(source.getEnabled());
    return target;
  }

  /**
   * Copy all fields from source to target
   * generated by @CopyFrom(Role) in {RoleResponse}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static RoleResponse copy(RoleResponse target, Role source) {
    target.setEnabled(source.getEnabled());
    target.setName(source.getName());
    target.setDescription(source.getDescription());
    target.setCode(source.getCode());
    target.setId(source.getId());
    return target;
  }

  /**
   * Copy all fields from source to target
   * generated by @CopyFrom(Permission) in {PermissionResponse}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static PermissionResponse copy(PermissionResponse target, Permission source) {
    target.setId(source.getId());
    target.setSystemCode(source.getSystemCode());
    target.setCode(source.getCode());
    target.setSystemId(source.getSystemId());
    target.setName(source.getName());
    target.setMenuPath(source.getMenuPath());
    return target;
  }

  /**
   * Copy all fields from source to target
   * generated by @CopyFrom(BusinessSystem) in {SystemResponse}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static SystemResponse copy(SystemResponse target, BusinessSystem source) {
    target.setCode(source.getCode());
    target.setName(source.getName());
    target.setId(source.getId());
    target.setBaseUrl(source.getBaseUrl());
    target.setEnabled(source.getEnabled());
    return target;
  }

  /**
   * Copy all fields from source to target
   * generated by @CopyTo(UserAccount) in {CreateUserRequest}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static UserAccount copy(UserAccount target, CreateUserRequest source) {
    target.setUsername(source.getUsername());
    target.setDisplayName(source.getDisplayName());
    return target;
  }

  /**
   * Copy non-null fields from source to target
   * generated by @CopyNotNullTo(UserAccount) in {UpdateUserRequest}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static UserAccount copyNotNull(UserAccount target, UpdateUserRequest source) {
    if (null != source.getEnabled()) {
      target.setEnabled(source.getEnabled());
    }
    if (null != source.getDisplayName()) {
      target.setDisplayName(source.getDisplayName());
    }
    return target;
  }
}
